#Requires -Version 5.1 <# .SYNOPSIS Compares installed software/agents between SF and TB VMs using PowerCLI Invoke-VMScript (VMware Tools) only. No WinRM, no SSH. .DESCRIPTION Assumes you are already connected to both vCenters (Connect-VIServer). Reads VM names from two plain text files (one per OS family). For each name, the script checks which vCenter (SF or TB) actually owns that VM and classifies it automatically - you do not tag names by site yourself. Inventory is collected entirely through Invoke-VMScript over VMware Tools, so no WinRM/PSRemoting and no SSH session is ever opened. Output: CSV files always, an Excel workbook if the ImportExcel module is available, an error log, and a console summary of processed / succeeded / failed VMs. .PARAMETER SFvCenterServer Name or IP of the already-connected SF vCenter (must match $global:DefaultVIServers). .PARAMETER TBvCenterServer Name or IP of the already-connected TB vCenter. .PARAMETER WindowsVMListFile Path to a text file with one Windows VM name per line. Lines starting with # are ignored. .PARAMETER LinuxVMListFile Path to a text file with one Linux VM name per line. Lines starting with # are ignored. .PARAMETER WindowsGuestCredential Credential used for Invoke-VMScript against every Windows VM (SF and TB). .PARAMETER SFLinuxGuestCredential Credential used for Invoke-VMScript against SF Linux VMs. Falls back to -LinuxGuestCredential if not supplied. .PARAMETER TBLinuxGuestCredential Credential used for Invoke-VMScript against TB Linux VMs. Falls back to -LinuxGuestCredential if not supplied. .EXAMPLE $winCred = Get-Credential -Message "Windows guest admin" $sfLinCred = Get-Credential -Message "SF Linux guest admin" $tbLinCred = Get-Credential Get-WindowsGuestSoftware { param([Parameter(Mandatory)]$VM, [Parameter(Mandatory)][System.Management.Automation.PSCredential]$Credential, [Parameter(Mandatory)][string]$Site) $psScript = @' $paths = SilentlyContinue | Where-Object { $_.Name -eq "VMware Tools" } | Select-Object -ExpandProperty Version -First 1) $kernel = (Get-CimInstance Win32_OperatingSystem).Version [PSCustomObject]@{ Software = $sw; ToolsVersion = $tools; OSVersion = $kernel } | ConvertTo-Json -Depth 4 -Compress '@ $raw = Invoke-GuestScriptSafe -VM $VM -ScriptText $psScript -ScriptType Powershell -Credential $Credential if (-not $raw) { throw "Empty result from guest script." } $parsed = $raw | ConvertFrom-Json $swList = @($parsed.Software) $agents = $swList | Where-Object { Get-AgentCategory $_.DisplayName } | ForEach-Object { [PSCustomObject]@{ VMName = $VM.Name; Site = $Site; Agent = $_.DisplayName; Version = $_.DisplayVersion; Category = Get-AgentCategory $_.DisplayName } } [PSCustomObject]@{ VMName = $VM.Name; Site = $Site; OSFamily = 'Windows' ToolsVersion = $parsed.ToolsVersion; OSVersion = $parsed.OSVersion SoftwareList = $swList; AgentsFound = $agents } } function Get-LinuxGuestSoftware { param([Parameter(Mandatory)]$VM, [Parameter(Mandatory)][System.Management.Automation.PSCredential]$Credential, [Parameter(Mandatory)][string]$Site) $bashScript = @' KERNEL=$(uname -r) if command -v vmware-toolbox-cmd >/dev/null 2>&1; then TOOLS=$(vmware-toolbox-cmd -v 2>/dev/null) elif command -v dpkg-query >/dev/null 2>&1; then TOOLS=$(dpkg-query -W -f='${Version}' open-vm-tools 2>/dev/null) elif command -v rpm >/dev/null 2>&1; then TOOLS=$(rpm -q --qf '%{VERSION}' open-vm-tools 2>/dev/null) else TOOLS="" fi if command -v dpkg-query >/dev/null 2>&1; then PKGS=$(dpkg-query -W -f='${Package}|${Version}\n' 2>/dev/null) elif command -v rpm >/dev/null 2>&1; then PKGS=$(rpm -qa --qf '%{NAME}|%{VERSION}\n' 2>/dev/null) else PKGS="" fi echo "KERNEL_START" echo "$KERNEL" echo "TOOLS_START" echo "$TOOLS" echo "PKGS_START" echo "$PKGS" '@ $raw = Invoke-GuestScriptSafe -VM $VM -ScriptText $bashScript -ScriptType Bash -Credential $Credential if (-not $raw) { throw "Empty result from guest script." } $lines = (($raw -replace "`r", '') -split "`n") | ForEach-Object { $_.Trim() } $kernelIdx = [array]::IndexOf($lines, 'KERNEL_START') $toolsIdx = [array]::IndexOf($lines, 'TOOLS_START') $pkgsIdx = [array]::IndexOf($lines, 'PKGS_START') if ($kernelIdx -lt 0 -or $toolsIdx -lt 0 -or $pkgsIdx -lt 0) { $preview = $raw.Substring(0, [Math]::Min(500, $raw.Length)) -replace "`r`n|`r|`n", ' | ' throw "Unexpected guest script output format. Raw output was: [$preview]" } $kernel = $lines[$kernelIdx + 1] $tools = $lines[$toolsIdx + 1] $pkgList = $lines[($pkgsIdx + 1)..($lines.Count - 1)] | Where-Object { $_ -match '\|' } | ForEach-Object { $parts = $_ -split '\|', 2 [PSCustomObject]@{ DisplayName = $parts[0]; DisplayVersion = $parts[1] } } $agents = $pkgList | Where-Object { Get-AgentCategory $_.DisplayName } | ForEach-Object { [PSCustomObject]@{ VMName = $VM.Name; Site = $Site; Agent = $_.DisplayName; Version = $_.DisplayVersion; Category = Get-AgentCategory $_.DisplayName } } [PSCustomObject]@{ VMName = $VM.Name; Site = $Site; OSFamily = 'Linux' ToolsVersion = $tools; OSVersion = $kernel SoftwareList = $pkgList; AgentsFound = $agents } } function Compare-SoftwareInventory { param($SFResults, $TBResults) $sfSw = $SFResults | ForEach-Object { $_.SoftwareList } $tbSw = $TBResults | ForEach-Object { $_.SoftwareList } $sfNames = $sfSw.DisplayName | Sort-Object -Unique $tbNames = $tbSw.DisplayName | Sort-Object -Unique $common = $sfNames | Where-Object { $_ -in $tbNames } $onlySF = $sfNames | Where-Object { $_ -notin $tbNames } $onlyTB = $tbNames | Where-Object { $_ -notin $sfNames } $rows = foreach ($name in $common) { $sfVer = ($sfSw | Where-Object DisplayName -eq $name | Select-Object -ExpandProperty DisplayVersion -Unique) -join ';' $tbVer = ($tbSw | Where-Object DisplayName -eq $name | Select-Object -ExpandProperty DisplayVersion -Unique) -join ';' [PSCustomObject]@{ SoftwareName = $name; Status = 'Common'; SF_Version = $sfVer; TB_Version = $tbVer; VersionMismatch = ($sfVer -ne $tbVer) } } $rows += $onlySF | ForEach-Object { $ver = ($sfSw | Where-Object DisplayName -eq $_ | Select-Object -ExpandProperty DisplayVersion -Unique) -join ';' [PSCustomObject]@{ SoftwareName = $_; Status = 'OnlyInSF'; SF_Version = $ver; TB_Version = 'Not Installed'; VersionMismatch = 'Not Installed' } } $rows += $onlyTB | ForEach-Object { $ver = ($tbSw | Where-Object DisplayName -eq $_ | Select-Object -ExpandProperty DisplayVersion -Unique) -join ';' [PSCustomObject]@{ SoftwareName = $_; Duration: $duration min. Succeeded: $succeeded Failed: $failed =====" Write-Host "" Write-Host "===== SUMMARY =====" -ForegroundColor Cyan Write-Host "Total VMs targeted : $totalTargets" Write-Host "Succeeded : $succeeded" -ForegroundColor Green Write-Host "Failed : $failed" -ForegroundColor $(if ($failed -gt 0) { 'Red' } else { 'Green' }) if ($ErrorLog.Count -gt 0) { Write-Host "Failure reasons:" -ForegroundColor Yellow $ErrorLog | ForEach-Object { Write-Host (" - [{0}] {1}: {2}" -f $_.Stage, $_.Object, $_.Reason) -ForegroundColor Yellow } } Write-Host "Reports : $OutputPath" -ForegroundColor Cyan
